// validationSchema.ts
import { z } from 'zod';

export const userSchema = z.object({
  name: z
    .string()
    .min(2, { message: 'Name must be at least 2 characters' })
    .max(50, { message: 'Name must be at most 50 characters' })
    .regex(/[A-Z]/, { message: 'Name must contain at least one uppercase letter' }),

  email: z
    .string()
    .email({ message: 'Invalid email address' }),

  password: z
    .string()
    .min(8, { message: 'Password must be at least 8 characters' })
    .regex(/[a-z]/, { message: 'Must include lowercase letter' })
    .regex(/[A-Z]/, { message: 'Must include uppercase letter' })
    .regex(/[0-9]/, { message: 'Must include a number' })
    .regex(/[@$!%*?&#]/, { message: 'Must include a special character' }),
});

// someComponent.ts
import { userSchema } from './validationSchema';

try {
  const parsed = userSchema.parse({
    name: 'Rupesh',
    email: 'rupesh@example.com',
    password: 'Telcy@123',
  });
  console.log('✅ Valid input:', parsed);
} catch (err) {
  console.log('❌ Validation errors:', err.errors);
}
